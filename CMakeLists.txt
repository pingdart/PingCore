cmake_minimum_required(VERSION 3.20)

# CMake initializes Apple deployment defaults during the first project() call.
# Explicit settings and toolchains retain ownership of their target platform.
if (CMAKE_HOST_APPLE AND (NOT CMAKE_SYSTEM_NAME OR CMAKE_SYSTEM_NAME STREQUAL "Darwin")
        AND NOT CMAKE_TOOLCHAIN_FILE AND "$ENV{CMAKE_TOOLCHAIN_FILE}" STREQUAL ""
        AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET AND "$ENV{MACOSX_DEPLOYMENT_TARGET}" STREQUAL "")
    set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS deployment target")
endif ()

project(pingcore VERSION 0.5.0 LANGUAGES NONE)

if (CMAKE_SYSTEM_NAME STREQUAL "iOS" AND NOT CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "15.0" CACHE STRING "Minimum iOS deployment target" FORCE)
endif ()

if (NOT PINGCORE_LIBRARY_GRAPH_ONLY)
    enable_language(CXX)
    if (APPLE)
        enable_language(OBJCXX)
    endif ()
endif ()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/lib")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/lib")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/bin")

if (PROJECT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR AND NOT ANDROID)
    set(PINGCORE_DEFAULT_BUILD_EXAMPLES ON)
    if (EMSCRIPTEN OR IOS)
        set(PINGCORE_DEFAULT_BUILD_TESTS OFF)
        set(PINGCORE_DEFAULT_BUILD_CLI OFF)
    else ()
        set(PINGCORE_DEFAULT_BUILD_TESTS ON)
        set(PINGCORE_DEFAULT_BUILD_CLI ON)
    endif ()
else ()
    set(PINGCORE_DEFAULT_BUILD_TESTS OFF)
    set(PINGCORE_DEFAULT_BUILD_EXAMPLES OFF)
    set(PINGCORE_DEFAULT_BUILD_CLI OFF)
endif ()

if (EMSCRIPTEN OR IOS OR ANDROID)
    set(PINGCORE_DEFAULT_BUILD_SHARED OFF)
else ()
    set(PINGCORE_DEFAULT_BUILD_SHARED ON)
endif ()

option(PINGCORE_BUILD_SHARED "Build the PingCore shared library" ${PINGCORE_DEFAULT_BUILD_SHARED})
option(PINGCORE_BUILD_STATIC "Build the PingCore static library" ON)
option(PINGCORE_BUILD_TESTS "Build PingCore tests" ${PINGCORE_DEFAULT_BUILD_TESTS})
option(PINGCORE_BUILD_TESTING_LIBRARY "Build the public windowless UI testing library" ${PINGCORE_DEFAULT_BUILD_TESTS})
option(PINGCORE_BUILD_TESTING_SMOKE_TESTS "Build windowless UI testing smoke tests for the current platform" OFF)
option(PINGCORE_BUILD_EXAMPLES "Build PingCore examples" ${PINGCORE_DEFAULT_BUILD_EXAMPLES})
option(PINGCORE_BUILD_CLI "Build the PingCore command-line tool" ${PINGCORE_DEFAULT_BUILD_CLI})
option(PINGCORE_ENABLE_PROFILING "Compile internal runtime performance diagnostics" OFF)
if (WIN32)
    option(PINGCORE_WINDOWS_7_COMPAT "Build the Windows backend for Windows 7 SP1 with Platform Update" OFF)
endif ()

set(PINGCORE_PROJECT_DIR "${PROJECT_SOURCE_DIR}")
set(PINGCORE_PUBLIC_INCLUDE_DIR "${PINGCORE_PROJECT_DIR}/include")
set(PINGCORE_PUBLIC_HEADER_DIR "${PINGCORE_PUBLIC_INCLUDE_DIR}/pingcore")
set(PINGCORE_SHARED_LIB_NAME "${PROJECT_NAME}")
set(PINGCORE_STATIC_LIB_NAME "${PROJECT_NAME}_static")

include(cmake/PingCoreBuild.cmake)
pingcore_configure_targets()
include(cmake/PingCoreApp.cmake)
include(cmake/PingCoreWindowsInstaller.cmake)

if (PINGCORE_LIBRARY_GRAPH_ONLY)
    return()
endif ()

if (PINGCORE_BUILD_TESTING_LIBRARY)
    include(cmake/PingCoreTesting.cmake)
endif ()
if (PINGCORE_BUILD_TESTING_SMOKE_TESTS)
    if (NOT PINGCORE_BUILD_TESTING_LIBRARY)
        message(FATAL_ERROR "PingCore testing smoke tests require PINGCORE_BUILD_TESTING_LIBRARY=ON")
    endif ()
    add_subdirectory(tests/platform/testing)
endif ()

if (PINGCORE_BUILD_CLI)
    add_subdirectory(tools/pingcore_cli)
endif ()

if (PROJECT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    pingcore_install_libraries()
    if ((NOT ANDROID OR PINGCORE_BUILD_CLI) AND NOT IOS AND NOT EMSCRIPTEN)
        include(cmake/PingCoreSdk.cmake)
    endif ()
endif ()

if (PINGCORE_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif ()

if (PINGCORE_BUILD_EXAMPLES)
    add_subdirectory(examples)
endif ()
