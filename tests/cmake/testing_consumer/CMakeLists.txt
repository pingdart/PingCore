cmake_minimum_required(VERSION 3.20)
project(pingcore_testing_consumer LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
if (PINGCORE_TEST_PRELOAD_CORE)
    find_package(PingCore CONFIG REQUIRED)
endif ()
if (PINGCORE_TEST_LINK_FORM)
    find_package(PingCore CONFIG REQUIRED COMPONENTS testing ${PINGCORE_TEST_LINK_FORM})
else ()
    find_package(PingCore CONFIG REQUIRED COMPONENTS testing)
endif ()
get_target_property(testing_core PingCore::testing INTERFACE_LINK_LIBRARIES)
if (PINGCORE_TEST_LINK_FORM STREQUAL "shared" AND NOT testing_core STREQUAL "PingCore::pingcore")
    message(FATAL_ERROR "PingCore testing ignored the requested shared core")
elseif (PINGCORE_TEST_LINK_FORM STREQUAL "static" AND NOT testing_core STREQUAL "PingCore::pingcore_static")
    message(FATAL_ERROR "PingCore testing ignored the requested static core")
endif ()
find_package(PingCore CONFIG REQUIRED COMPONENTS testing)
if (PINGCORE_TEST_CONFLICT_LINK_FORM)
    find_package(PingCore CONFIG REQUIRED COMPONENTS testing ${PINGCORE_TEST_CONFLICT_LINK_FORM})
endif ()
add_executable(testing_consumer main.cpp)
target_link_libraries(testing_consumer PRIVATE PingCore::testing)
if (WIN32 AND testing_core STREQUAL "PingCore::pingcore")
    add_custom_command(TARGET testing_consumer POST_BUILD
            COMMAND "${CMAKE_COMMAND}" -E copy_if_different
                    "$<TARGET_FILE:PingCore::pingcore>" "$<TARGET_FILE_DIR:testing_consumer>"
            VERBATIM)
endif ()
enable_testing()
add_test(NAME InstalledTestingConsumer COMMAND testing_consumer)
